<?php
/**
 * Banc de mesure (phase 3, tâche 20) : base de données d'appoint, sans connexion.
 * WordPress est amorcé en SHORTINIT pour charger son vrai code HTTP (WP_Http,
 * Requests, transport cURL) ; aucune requête SQL n'est utile à la mesure.
 */
class G2rd_Mesure_Wpdb extends wpdb {
	public function __construct() {
		$this->show_errors = false;
		$this->ready       = true;
	}
	public function db_connect( $allow_bail = true ) {
		return false;
	}
	public function query( $query ) {
		return false;
	}
	public function get_results( $query = null, $output = OBJECT ) {
		return array();
	}
	public function get_row( $query = null, $output = OBJECT, $y = 0 ) {
		return null;
	}
	public function get_var( $query = null, $x = 0, $y = 0 ) {
		return null;
	}
	public function get_col( $query = null, $x = 0 ) {
		return array();
	}
	public function _real_escape( $data ) {
		return is_scalar( $data ) ? addslashes( (string) $data ) : '';
	}
}
$wpdb = new G2rd_Mesure_Wpdb();
